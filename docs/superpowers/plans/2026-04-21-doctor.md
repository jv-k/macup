# `--doctor` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a root-level `--doctor` flag (with optional `--json`) that runs five diagnostic sections (environment, config, plugins, data integrity, shell integration) and prints a sectioned health report. Exits 0 if clean, 1 on any error; warnings never fail.

**Architecture:** One pure report type (`CheckResult` / `Section`) plus one renderer pair (text + JSON), shared by five check modules. Each check module owns one section and has one exported `check(deps)` function. `src/commands/doctor.ts` orchestrates them in parallel and dispatches to the renderers. `src/cli.ts` wires the flag into the existing root command, matching the `--config` / `--cleanup` / `--restore` pattern.

**Tech Stack:** TypeScript ESM, vitest, citty, execa (via `ExecRunner`), picocolors, `@clack/prompts` (unchanged).

**Spec:** [docs/superpowers/specs/2026-04-21-doctor-design.md](../specs/2026-04-21-doctor-design.md)

```ts
import * as log from '../../ui/log';

export type Level = 'ok' | 'warn' | 'error';

export interface CheckResult {
  readonly level: Level;
  readonly label: string;
  readonly detail?: string;
  /** Optional actionable next-step line, rendered dim under the main line. */
  readonly hint?: string;
  /** 0 = top-level row, 1 = sub-row (rendered with ↳ prefix). Default 0. */
  readonly indent?: number;
}

export interface Section {
  readonly title: string;
  readonly results: readonly CheckResult[];
}

export interface Summary {
  readonly ok: number;
  readonly warn: number;
  readonly error: number;
  readonly exitCode: 0 | 1;
}

export function summarize(sections: readonly Section[]): Summary {
  let ok = 0;
  let warn = 0;
  let err = 0;
  for (const s of sections) {
    for (const r of s.results) {
      if (r.level === 'ok') ok++;
      else if (r.level === 'warn') warn++;
      else err++;
    }
  }
  return { ok, warn, error: err, exitCode: err > 0 ? 1 : 0 };
}

export function renderJson(sections: readonly Section[]): string {
  const summary = summarize(sections);
  return JSON.stringify({ summary, sections }, null, 2);
}

const SYM: Record<Level, string> = {
  ok: '✔',
  warn: '!',
  error: '✖',
};

function formatLine(r: CheckResult, labelPad: number): string {
  const detail = r.detail ? `  ${r.detail}` : '';
  // Sub-rows render as a "nested note" — no symbol, no padding, ↳ carries the
  // visual continuation. Top-level rows get the ok/warn/error symbol and a
  // padded label so the details align column-wise.
  const line = r.indent === 1
    ? `    ↳ ${r.label}${detail}`
    : `  ${SYM[r.level]} ${r.label.padEnd(labelPad)}${detail}`;
  return r.hint ? `${line}\n    ↳ ${r.hint}` : line;
}

export function renderText(sections: readonly Section[]): string {
  const nonEmpty = sections.filter((s) => s.results.length > 0);
  const lines: string[] = [];

  lines.push('');
  lines.push(log.header('macup doctor'));
  lines.push('');

  for (const section of nonEmpty) {
    const labelPad = Math.max(
      12,
      ...section.results.map((r) => r.label.length),
    );
    lines.push(log.header(section.title));
    for (const r of section.results) {
      lines.push(formatLine(r, labelPad));
    }
    lines.push('');
  }

  const s = summarize(nonEmpty);
  const summaryLine = `SUMMARY: ${s.ok} ok, ${s.warn} warning${s.warn === 1 ? '' : 's'}, ${s.error} error${s.error === 1 ? '' : 's'}`;
  if (s.error > 0) lines.push(log.error(summaryLine));
  else if (s.warn > 0) lines.push(log.warning(summaryLine));
  else lines.push(log.success(summaryLine));

  return lines.join('\n');
}
```

```ts
import type { CheckResult, Section } from '../report';

export interface EnvironmentDeps {
  readonly platform: string;
  readonly release: string;
  readonly arch: string;
  readonly nodeVersion: string;
  readonly shellEnv: string | undefined;
  readonly macupVersion: string;
}

function parseNodeMajor(version: string): number | null {
  const m = /^v(\d+)\./.exec(version);
  return m?.[1] ? Number.parseInt(m[1], 10) : null;
}

export async function check(deps: EnvironmentDeps): Promise<Section> {
  const results: CheckResult[] = [];

  results.push({
    level: 'ok',
    label: 'macOS',
    detail: `${deps.platform} ${deps.release} (${deps.arch})`,
  });

  const shellName = deps.shellEnv?.split('/').pop();
  if (shellName) {
    results.push({ level: 'ok', label: 'Shell', detail: shellName });
  } else {
    results.push({
      level: 'warn',
      label: 'Shell',
      detail: 'unknown (SHELL env var is unset)',
    });
  }

  const nodeMajor = parseNodeMajor(deps.nodeVersion);
  if (nodeMajor !== null && nodeMajor >= 20) {
    results.push({ level: 'ok', label: 'Node', detail: `${deps.nodeVersion} (>= 20 required)` });
  } else {
    results.push({
      level: 'error',
      label: 'Node',
      detail: deps.nodeVersion,
      hint: 'upgrade to Node >= 20',
    });
  }

  results.push({ level: 'ok', label: 'macup', detail: `v${deps.macupVersion}` });

  return { title: 'Environment', results };
}
```

```ts
import type { CheckResult, Section } from '../report';

export interface ConfigDeps {
  readonly configDir: string;
  readonly configDirExists: boolean;
  readonly configDirWritable: boolean;
  readonly applistPath: string;
  readonly applistExists: boolean;
  readonly applistBytes: number;
  readonly applistEntryCount: number;
  readonly applistListCount: number;
  readonly applistError: string | undefined;
  readonly backupDir: string;
  readonly backupDirExists: boolean;
  readonly backupDirWritable: boolean;
  readonly backupCount: number;
  readonly backupTotalBytes: number;
}

function kb(bytes: number): string {
  return `${(bytes / 1024).toFixed(1)}kb`;
}

export async function check(deps: ConfigDeps): Promise<Section> {
  const results: CheckResult[] = [];

  if (!deps.configDirExists) {
    results.push({
      level: 'warn',
      label: 'Config dir',
      detail: `${deps.configDir} (not yet created)`,
    });
  } else if (!deps.configDirWritable) {
    results.push({
      level: 'warn',
      label: 'Config dir',
      detail: `${deps.configDir} (not writable)`,
      hint: `chmod u+w ${deps.configDir}`,
    });
  } else {
    results.push({ level: 'ok', label: 'Config dir', detail: deps.configDir });
  }

  if (deps.applistError) {
    results.push({
      level: 'error',
      label: 'applist.yaml',
      detail: deps.applistError,
    });
  } else if (!deps.applistExists) {
    results.push({
      level: 'warn',
      label: 'applist.yaml',
      detail: 'not created yet — run `macup <plugin> add <name>` to start tracking',
    });
  } else {
    results.push({
      level: 'ok',
      label: 'applist.yaml',
      detail: `${kb(deps.applistBytes)}, ${deps.applistListCount} lists (${deps.applistEntryCount} entries), schema OK`,
    });
  }

  if (!deps.backupDirExists) {
    results.push({
      level: 'ok',
      label: 'Backup dir',
      detail: `${deps.backupDir} (no backups yet)`,
    });
  } else if (!deps.backupDirWritable) {
    results.push({
      level: 'warn',
      label: 'Backup dir',
      detail: `${deps.backupDir} (not writable)`,
    });
  } else {
    results.push({
      level: 'ok',
      label: 'Backup dir',
      detail: `${deps.backupDir} (${deps.backupCount} backups, ${kb(deps.backupTotalBytes)})`,
    });
  }

  return { title: 'Config', results };
}
```

```ts
import type { Plugin, PluginContext } from '../../../plugins/types';
import type { CheckResult, Section } from '../report';

export interface PluginsDeps {
  readonly plugins: readonly Plugin[];
  readonly onPath: (binary: string) => boolean;
  readonly probeVersion: (plugin: Plugin) => Promise<string>;
  readonly ctx: PluginContext;
  /** Default 15000. */
  readonly probeTimeoutMs?: number;
}

async function withTimeout<T>(
  promise: Promise<T>,
  ms: number,
  label: string,
): Promise<T> {
  let handle: NodeJS.Timeout | undefined;
  try {
    return await Promise.race([
      promise,
      new Promise<never>((_, reject) => {
        handle = setTimeout(
          () => reject(new Error(`${label} timed out after ${ms}ms`)),
          ms,
        );
      }),
    ]);
  } finally {
    if (handle) clearTimeout(handle);
  }
}

export async function check(deps: PluginsDeps): Promise<Section> {
  const timeoutMs = deps.probeTimeoutMs ?? 15_000;
  const results: CheckResult[] = [];

  const shown = deps.plugins.filter((p) => p.manifest.id !== 'all');

  for (const plugin of shown) {
    const missingBin = plugin.manifest.requires.find((b) => !deps.onPath(b));
    if (missingBin) {
      results.push({
        level: 'warn',
        label: plugin.manifest.id,
        detail: `${missingBin} not on PATH — plugin disabled`,
      });
      continue;
    }

    let versionDetail: string;
    try {
      versionDetail = await deps.probeVersion(plugin);
    } catch (err) {
      versionDetail = err instanceof Error ? err.message : String(err);
    }
    results.push({
      level: 'ok',
      label: plugin.manifest.id,
      detail: versionDetail,
    });

    // Deep probe: call plugin.list() to confirm the integration works.
    try {
      const statuses = await withTimeout(
        plugin.list(deps.ctx, { onlyOutdated: true }),
        timeoutMs,
        `${plugin.manifest.id} list probe`,
      );
      results.push({
        level: 'ok',
        label: 'list probe',
        detail: `${statuses.length} tracked`,
        indent: 1,
      });
    } catch (err) {
      results.push({
        level: 'warn',
        label: 'list probe',
        detail: err instanceof Error ? err.message : String(err),
        indent: 1,
      });
    }
  }

  return { title: 'Plugins', results };
}
```

```ts
import type { Plugin, PluginContext } from '../../../plugins/types';
import type { CheckResult, Section } from '../report';

export interface PinEntry {
  readonly pluginId: string;
  readonly name: string;
  readonly version: string;
}

export interface SkipEntry {
  readonly pluginId: string;
  readonly name: string;
}

export interface DataIntegrityDeps {
  readonly plugins: readonly Plugin[];
  readonly trackedFor: (plugin: Plugin) => readonly string[];
  readonly pins: readonly PinEntry[];
  readonly skips: readonly SkipEntry[];
  readonly ctx: PluginContext;
}

export async function check(deps: DataIntegrityDeps): Promise<Section> {
  const results: CheckResult[] = [];

  // Build a map of pluginId -> Set(known names from plugin.list() AND
  // from the user's tracked list). A name known to either counts as
  // "resolvable" — tracked-but-not-installed is the install command's
  // concern, not doctor's.
  const knownByPlugin = new Map<string, Set<string>>();
  let listedTotal = 0;
  let listCount = 0;

  for (const plugin of deps.plugins) {
    if (plugin.manifest.id === 'all') continue;
    if (plugin.manifest.configKeys.length === 0) continue;

    const tracked = deps.trackedFor(plugin);
    listedTotal += tracked.length;
    listCount += 1;

    try {
      const statuses = await plugin.list(deps.ctx, {});
      const known = new Set<string>();
      for (const s of statuses) known.add(s.ref.name);
      for (const t of tracked) known.add(t);
      knownByPlugin.set(plugin.manifest.id, known);

      for (const name of tracked) {
        const present = statuses.some((s) => s.ref.name === name);
        if (!present) {
          results.push({
            level: 'warn',
            label: 'Stale tracked entry',
            detail: `${plugin.manifest.id}:${name} is tracked but not installed`,
          });
        }
      }
    } catch {
      // Swallow and let the plugins-check section surface it — data-integrity
      // shouldn't cascade a single plugin's failure into error noise.
      knownByPlugin.set(plugin.manifest.id, new Set(tracked));
    }
  }

  // Headline row.
  results.unshift({
    level: results.length === 0 ? 'ok' : 'warn',
    label: 'Tracked packages',
    detail: `${listedTotal} entries across ${listCount} list${listCount === 1 ? '' : 's'}${
      results.length === 0 ? ', all resolvable' : ''
    }`,
  });

  for (const pin of deps.pins) {
    const known = knownByPlugin.get(pin.pluginId);
    if (!known || !known.has(pin.name)) {
      results.push({
        level: 'warn',
        label: 'Stale pin',
        detail: `${pin.pluginId}:${pin.name} pinned ${pin.version} but not in tracked list`,
      });
    }
  }

  for (const skip of deps.skips) {
    const known = knownByPlugin.get(skip.pluginId);
    if (!known || !known.has(skip.name)) {
      results.push({
        level: 'warn',
        label: 'Stale skip',
        detail: `${skip.pluginId}:${skip.name} skipped but not in tracked list`,
      });
    }
  }

  return { title: 'Data integrity', results };
}
```

```ts
import type { CheckResult, Section } from '../report';

export interface ShellIntegrationDeps {
  readonly shellEnv: string | undefined;
  readonly home: string;
  readonly pathExists: (path: string) => boolean;
}

type SupportedShell = 'zsh' | 'bash' | 'fish';

function detect(shellEnv: string | undefined): SupportedShell | null {
  const name = shellEnv?.split('/').pop();
  if (name === 'zsh' || name === 'bash' || name === 'fish') return name;
  return null;
}

function candidatesFor(shell: SupportedShell, home: string): string[] {
  if (shell === 'zsh') {
    return [
      `${home}/.zsh/completions/_macup`,
      '/usr/local/share/zsh/site-functions/_macup',
      '/opt/homebrew/share/zsh/site-functions/_macup',
    ];
  }
  if (shell === 'bash') {
    return [
      `${home}/.local/share/bash-completion/completions/macup`,
      '/usr/local/etc/bash_completion.d/macup',
      '/etc/bash_completion.d/macup',
    ];
  }
  return [`${home}/.config/fish/completions/macup.fish`];
}

function installHint(shell: SupportedShell, home: string): string {
  if (shell === 'zsh') {
    return `macup --completions=zsh >> ${home}/.zsh/completions/_macup`;
  }
  if (shell === 'bash') {
    return `macup --completions=bash >> ${home}/.local/share/bash-completion/completions/macup`;
  }
  return `macup --completions=fish > ${home}/.config/fish/completions/macup.fish`;
}

export async function check(deps: ShellIntegrationDeps): Promise<Section> {
  const results: CheckResult[] = [];
  const shell = detect(deps.shellEnv);

  if (!shell) {
    results.push({
      level: 'warn',
      label: 'Completions',
      detail: `unsupported shell (${deps.shellEnv ?? 'unset'}) — zsh, bash, fish only`,
    });
    return { title: 'Shell integration', results };
  }

  const candidates = candidatesFor(shell, deps.home);
  const found = candidates.find((p) => deps.pathExists(p));

  if (found) {
    results.push({
      level: 'ok',
      label: 'Completions',
      detail: `${shell} — installed at ${found}`,
    });
  } else {
    results.push({
      level: 'warn',
      label: 'Completions',
      detail: `${shell} — not installed`,
      hint: installHint(shell, deps.home),
    });
  }

  return { title: 'Shell integration', results };
}
```

```ts
function isWritable(path: string): boolean {
  try {
    accessSync(path, constants.W_OK);
    return true;
  } catch {
    return false;
  }
}

function fileBytes(path: string): number {
  try {
    return statSync(path).size;
  } catch {
    return 0;
  }
}

function sumBackupBytes(dir: string): { count: number; bytes: number } {
  try {
    const entries = readdirSync(dir);
    let bytes = 0;
    for (const name of entries) {
      bytes += fileBytes(join(dir, name));
    }
    return { count: entries.length, bytes };
  } catch {
    return { count: 0, bytes: 0 };
  }
}

async function probeVersion(exec: ExecaExecRunner, plugin: Plugin): Promise<string> {
  const primary = plugin.manifest.requires[0];
  if (!primary) return plugin.manifest.id;
  const args = VERSION_FLAG_OVERRIDES[primary] ?? ['--version'];
  try {
    const r = await exec.run(primary, args);
    const firstLine = (r.stdout || r.stderr).split('\n')[0]?.trim() ?? '';
    return firstLine.length > 0 ? firstLine : `${primary} (no version output)`;
  } catch (err) {
    return err instanceof Error ? err.message : String(err);
  }
}

      pluginsCheck({
        plugins: BUILTIN_PLUGINS,
        onPath: (bin) => isOnPath(bin),
        probeVersion: (plugin) => probeVersion(exec, plugin),
        ctx,
      }),
      dataCheck({
        plugins: BUILTIN_PLUGINS,
        trackedFor,
        pins,
        skips,
        ctx,
      }),
      shellCheck({
        shellEnv: process.env.SHELL,
        home: homedir(),
        pathExists: (p) => existsSync(p),
      }),
    ]);

  const sections: readonly Section[] = [
    envSection,
    configSection,
    pluginsSection,
    dataSection,
    shellSection,
  ];

  if (opts.json) {
    console.log(renderJson(sections));
  } else {
    console.log(renderText(sections));
  }

  return summarize(sections).exitCode;
}
```

## Risks / watch-outs

- **`probeVersion` on `softwareupdate`:** `softwareupdate` has no `--version`. We use `--help` and capture the first line of stderr (it prints a usage banner). Sufficient for a sanity check that the binary runs.
- **Parallel `plugin.list()` calls:** each plugin is I/O-heavy; running five in parallel could spike CPU or saturate network (brew outdated hits the registry). Acceptable for a diagnostic that runs on-demand; document if users complain.
- **`plugin.list({})` on a fresh install:** when applist.yaml doesn't exist yet, the data-integrity section correctly reports zero tracked entries (the "0 entries across 0 lists, all resolvable" edge case renders as `ok`).
- **Citty arg ordering:** the `--json` flag at the root level is a boolean; if a user invokes `macup --json list` thinking it'll chain to a subcommand, citty will treat `list` as a rawArg. The branch in `run()` only fires on `--doctor`; other paths are unaffected.
- **Test environment variance:** the regression test spawns `dist/cli.mjs` against the real host's brew/npm/etc. A CI box without `brew` will cause `pnpm outdated` under `brew outdated` to fail — but we swallow those failures at the plugin/data-integrity level as warnings, so exit stays 0. The test only checks structure, not content.
